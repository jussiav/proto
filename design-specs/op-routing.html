Design initiative · success page, decision page

OP routing

A pilot: AutoVex suggests car insurance from OP at moments when the seller has nothing left to do with us. One block, one button out to OP's insurance page, and no change to the selling flow.

Status: in discussion · 2 variants, 2 placements November 2026

Part 1 · for the team and OP

Summary

A seller selling their car will soon need insurance for the next one. We suggest OP's car insurance where the seller has nothing left to do with AutoVex, and send them to OP's own page for a quote.

Two placements: the success page after submitting the listing, and the decision page after the seller has accepted an offer or the offers were rejected.

Two designs: variant 3, AutoVex's own block, whose design OP has accepted, and variant 4, OP's own banner built as HTML, which is still being discussed. Both run on the two pages as one VWO front-end test (part 2).

Emails are not part of this initiative. Marketing handles any email placement.

It must not cost the selling flow anything. The block appears only where nothing is asked of the seller, below AutoVex's own message on the page, and never replaces or delays a step.

Design accepted by OP

  • One button, filled with OP's orange, 56px tall: “Laske vakuutustarjous OP:ssa” (calculate an insurance quote at OP). It opens OP's car insurance page in a new tab; the AutoVex page stays open.
  • The terms line under the button: the offer covers the first insurance period of new comprehensive insurance.
  • A coupon frame: 2px dashed light grey border, 12px corners, white fill, 20px padding, at most 448px wide.
  • AutoVex's fonts (Barlow, DM Sans); nothing to license.

Placements

1 — Success page

  • When: only where the seller has nothing left to do: the email is verified, and the listing is in review or published.
  • Never while email verification is pending (that is the seller's task), on a rejected listing, or when the photos are missing.
  • Where: 32px under the cream “Ota rennosti ja tarkkaile sähköpostiasi!” notice, in the 448px left column. In review, the Hotjar rating survey stays last, under the block.

Variant 3: in review · published. Variant 4: in review · published.

2 — Decision page

  • When: the seller has accepted an offer (the page says “Onneksi olkoon!”), or the offers were rejected (“Tarjoukset hylätty”), including a rejected final offer. The decision is made, so nothing is asked of the seller any more.
  • Not while an offer can still be accepted, on a first visit before “Katso tulokset”, with no offers, or when the offers expired.
  • Where: as its own section directly under the “Tarvitsetko apua?” banner, so AutoVex's own help comes first and the auction result follows. Centred, at most 448px wide.
  • Starting with these two states. Other decision page states may follow later.

Variant 3: after accepting · after rejecting. Variant 4: after accepting · after rejecting.

  • The OP Chevin Pro web font. The prototype loads it from op.fi's own servers (op.fi allows other sites to load these files). Production needs OP to confirm that autovex.fi may do the same, or a licence to host it: the brand package's licence (G-Type) forbids redistributing the font or serving it from a web server without a web-font licence. Without the font, the banner shows in Arial.
  • The target page. OP's car insurance page with OP's tracking code, one link per placement (B7).

With Legal

  • What the link carries to OP. Only OP's campaign code naming the placement; nothing about the seller or the car.
  • Does the block need a marketing label? Consumer law requires marketing to be recognisable as marketing and to show on whose behalf it is made; whether OP's name and logo are enough, or a label such as “Mainos” or “Yhteistyössä OP” is needed, is for Legal.

  function visible(el) {
    return !!el && el.getClientRects().length > 0;
  }

  // Funnel success screen. The cream notice with the link to the offers page
  // ("Katso ilmoitukseni") is drawn only for a verified seller whose ad is in
  // review (Review.vue) or published (Success.vue). The unverified, rejected
  // and photos-missing screens have no such notice.
  function successAnchor() {
    var notices = document.querySelectorAll('div.border-orange-300.rounded-lg');
    for (var i = 0; i < notices.length; i++) {
      if (visible(notices[i]) && notices[i].querySelector('a[href*="offers"]')) return notices[i];
    }
    return null;
  }

  // Decision page. The hero's own title names the state; the snippet reads
  // nothing else about the auction.
  function decisionAnchor() {
    var titles = document.querySelectorAll('h2.text-3xl.font-bold.text-black');
    var hero = null;
    for (var i = 0; i < titles.length; i++) {
      if (DECISION_TITLES.indexOf(titles[i].textContent.trim()) !== -1 && visible(titles[i])) { hero = titles[i]; break; }
    }
    if (!hero) return null;
    var offers = hero.closest('section');
    if (!offers) return null;
    var next = offers.nextElementSibling;
    while (next && next.id !== ID && !visible(next)) next = next.nextElementSibling;
    if (next && next.id === ID) next = null;
    // The "Tarvitsetko apua?" banner, when the page puts it right after the
    // offers (accepted, or every offer rejected).
    if (next && next.tagName === 'SECTION' && next.querySelector('h5')) return next;
    return offers;
  }

  function injectCss() {
    if (document.getElementById(ID + '-style')) return;
    var style = document.createElement('style');
    style.id = ID + '-style';
    style.textContent = CSS;
    document.head.appendChild(style);
  }

  function place(anchor, tag, cls, inner) {
    var existing = document.getElementById(ID);
    if (existing && existing.previousElementSibling === anchor) return;
    if (existing) existing.remove();
    injectCss();
    anchor.insertAdjacentHTML('afterend', '<' + tag + ' id="' + ID + '"' + (cls ? ' class="' + cls + '"' : '') + '>' + inner + '</' + tag + '>');
  }

  function sync() {
    // Finnish only, like the copy.
    if (/^en/i.test(document.documentElement.lang || '')) return remove();
    var anchor = successAnchor();
    if (anchor) return place(anchor, 'div', 'av-op-success', card(URL_SUCCESS));
    anchor = decisionAnchor();
    if (anchor) return place(anchor, 'section', '', '<div class="container mx-auto">' + card(URL_DECISION) + '</div>');
    remove();
  }

  function remove() {
    var existing = document.getElementById(ID);
    if (existing) existing.remove();
  }

  // Both pages are single-page apps that mount after load and re-render on
  // every action, so sync whenever the DOM changes. sync is idempotent.
  function start() {
    sync();
    new MutationObserver(sync).observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['class'] });
  }
  if (document.body) start();
  else document.addEventListener('DOMContentLoaded', start);
}());

  function visible(el) {

  // Funnel success screen. The cream notice with the link to the offers page
  // ("Katso ilmoitukseni") is drawn only for a verified seller whose ad is in

  // and photos-missing screens have no such notice.
  function successAnchor() {
    var notices = document.querySelectorAll('div.border-orange-300.rounded-lg');
    for (var i = 0; i < notices.length; i++) {
      if (visible(notices[i]) && notices[i].querySelector('a[href*="offers"]')) return notices[i];

  }

  // Decision page. The hero's own title names the state; the snippet reads
  // nothing else about the auction.
  function decisionAnchor() {
    var titles = document.querySelectorAll('h2.text-3xl.font-bold.text-black');
    var hero = null;
    for (var i = 0; i < titles.length; i++) {
      if (DECISION_TITLES.indexOf(titles[i].textContent.trim()) !== -1 && visible(titles[i])) { hero = titles[i]; break; }
    }
    if (!hero) return null;
    var offers = hero.closest('section');
    if (!offers) return null;
    var next = offers.nextElementSibling;
    while (next && next.id !== ID && !visible(next)) next = next.nextElementSibling;
    if (next && next.id === ID) next = null;
    // The "Tarvitsetko apua?" banner, when the page puts it right after the
    // offers (accepted, or every offer rejected).
    if (next && next.tagName === 'SECTION' && next.querySelector('h5')) return next;
    return offers;
  }

  function injectCss() {
    if (document.getElementById(ID + '-style')) return;
    var style = document.createElement('style');

  function place(anchor, tag, cls, inner) {
    var existing = document.getElementById(ID);
    if (existing && existing.previousElementSibling === anchor) return;
    if (existing) existing.remove();
    injectCss();
    anchor.insertAdjacentHTML('afterend', '<' + tag + ' id="' + ID + '"' + (cls ? ' class="' + cls + '"' : '') + '>' + inner + '</' + tag + '>');
  }

  function remove() {
    var existing = document.getElementById(ID);
    if (existing) existing.remove();
  }

  function sync() {
    // Finnish only, like the copy.
    if (/^en/i.test(document.documentElement.lang || '')) return remove();
    var anchor = successAnchor();
    if (anchor) return place(anchor, 'div', 'av-op-success', banner(URL_SUCCESS));
    anchor = decisionAnchor();
    if (anchor) return place(anchor, 'section', '', '<div class="container mx-auto">' + banner(URL_DECISION) + '</div>');
    remove();
  }

    sync();
    new MutationObserver(sync).observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['class'] });
  }
  if (document.body) start();
  else document.addEventListener('DOMContentLoaded', start);
